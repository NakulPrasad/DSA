# Min Stack

```java
class MinStack {
    Stack<Integer> normal = new Stack<>();
    Stack<Integer> min = new Stack<>();
    public MinStack() {
        
    }
    
    public void push(int val) {
        normal.push(val);
        if(min.isEmpty()) min.push(val);
        else
        min.push(Math.min(min.peek(), val));
    }
    
    public void pop() {
        if(!normal.isEmpty() && !min.isEmpty()){

        normal.pop();
        min.pop();
        }
    }
    
    public int top() {
        if(!normal.isEmpty())
        return normal.peek();
        return 0;
    }
    
    public int getMin() {
        if(!min.isEmpty())
        return min.peek();
        return 0;
    }
}
```
